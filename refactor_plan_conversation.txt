/// Single unified buffer type that works with any backend
pub struct MemoryBuffer {
    ptr: *mut u8,
    size: usize,
    info: BufferInfo,
    backend: Box<dyn Backend>,
}

pub struct BufferInfo {
    pub physical_address: Option<u64>,
    pub numa_node: u32,
    pub page_type: PageType,
    pub segments: Vec<SegmentInfo>,  // Empty for simple allocations
    pub memory_type: MemoryType,
    pub supports_remapping: bool,
}

#[derive(Debug, Clone, Copy)]
pub enum PageType {
    Regular(usize),      // 4KB
    Large(usize),        // 2MB  
    Huge(usize),         // 1GB
    Mixed(Vec<PageInfo>), // For segmented allocations
}

impl MemoryBuffer {
    pub fn as_mut_ptr(&self) -> *mut u8 { self.ptr }
    pub fn size(&self) -> usize { self.size }
    pub fn info(&self) -> &BufferInfo { &self.info }
    
    pub fn remap_memory_type(&mut self, new_type: MemoryType) -> Result<(), String> {
        if self.info.supports_remapping {
            self.backend.remap_memory_type(self.ptr as u64, new_type)?;
            self.info.memory_type = new_type;
            Ok(())
        } else {
            Err("This allocation doesn't support memory type remapping".into())
        }
    }
}

4. memory/allocator.rs - High-level allocation API
rust
pub struct MemoryAllocator {
    backend: Box<dyn Backend>,
    stats: AllocationStats,
}

pub struct AllocationConfig {
    pub size: usize,
    pub numa_node: Option<u32>,
    pub page_size: PageSizePreference,
    pub memory_type: MemoryType,
    pub zero_memory: bool,
    pub timeout_ms: u32,
}

pub enum PageSizePreference {
    Any,                    // Let backend decide
    Prefer(PageType),       // Prefer but fall back
    Require(PageType),      // Must have or fail
    Range { min: PageType, max: PageType },
}

impl MemoryAllocator {
    pub fn new(backend_type: BackendType) -> Result<Self, String> {
        let backend: Box<dyn Backend> = match backend_type {
            BackendType::Auto => Self::auto_detect_backend()?,
            BackendType::Windows { large_pages } => {
                Box::new(WindowsBackend { use_large_pages })
            }
            BackendType::Driver => {
                Box::new(DriverBackend::new()?)
            }
        };
        
        Ok(Self {
            backend,
            stats: AllocationStats::default(),
        })
    }
    
    pub fn allocate(&mut self, config: AllocationConfig) -> Result<MemoryBuffer, String> {
        let allocation = self.backend.allocate(&config)?;
        self.stats.record_allocation(&allocation);
        
        Ok(MemoryBuffer {
            ptr: allocation.ptr,
            size: allocation.size,
            info: allocation.info,
            backend: self.backend.clone(), // Need Arc for this
        })
    }
}
